cmake_minimum_required(VERSION 3.15)
project(raisin_d405_plugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Match the framework's Eigen allocation ABI when building outside its root.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT WIN32)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(D405_DEFAULT_MARCH "armv8.2-a+crypto+fp16+dotprod")
  else()
    set(D405_DEFAULT_MARCH "x86-64-v3")
  endif()
  set(RAISIN_MARCH "${D405_DEFAULT_MARCH}" CACHE STRING "Match the installed Raisin SDK -march")
  add_compile_options("-march=${RAISIN_MARCH}")
endif()
if(NOT COMMAND raisin_find_package)
  include(${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/raisin_helper.cmake)
endif()
raisin_find_package(raisin_plugin REQUIRED)
raisin_find_package(raisin_network REQUIRED)
raisin_find_package(realsense2 REQUIRED)
raisin_find_package(raisin_av1_encoder REQUIRED)
raisin_find_package(OpenCV REQUIRED)
raisin_find_package(yaml-cpp REQUIRED)
raisin_find_package(rayrai REQUIRED)
add_library(${PROJECT_NAME} SHARED src/d405_plugin.cpp)
target_link_libraries(${PROJECT_NAME} PRIVATE raisin_plugin raisin_network
  realsense2::realsense2 raisin_av1_encoder yaml-cpp rayrai ${OpenCV_LIBS})
target_include_directories(${PROJECT_NAME} PRIVATE ${OpenCV_INCLUDE_DIRS})
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY config/ DESTINATION config/${PROJECT_NAME}/config)
install(DIRECTORY resource/ DESTINATION resource/${PROJECT_NAME}/resource)
include(CTest)
if(BUILD_TESTING)
  add_executable(d405_rayrai_rgbd_check test/rayrai_rgbd_check.cpp)
  target_link_libraries(d405_rayrai_rgbd_check PRIVATE rayrai)
  add_test(NAME d405_rayrai_rgbd COMMAND d405_rayrai_rgbd_check)
  set_tests_properties(d405_rayrai_rgbd PROPERTIES TIMEOUT 30)
  add_executable(d405_nonreal_smoke test/nonreal_smoke.cpp)
  set_target_properties(d405_nonreal_smoke PROPERTIES ENABLE_EXPORTS TRUE)
  target_link_libraries(d405_nonreal_smoke PRIVATE raisin_plugin raisin_network ${CMAKE_DL_LIBS})
  add_dependencies(d405_nonreal_smoke ${PROJECT_NAME})
  add_test(NAME d405_nonreal COMMAND d405_nonreal_smoke
    "$<TARGET_FILE:${PROJECT_NAME}>"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../raisin_raipal/resource/urdf_raipal/urdf/raipal_stub-0_R.urdf"
    "${CMAKE_INSTALL_PREFIX}")
  set_tests_properties(d405_nonreal PROPERTIES TIMEOUT 30)
endif()
